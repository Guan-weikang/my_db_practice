### 4.1 配偶与子女查询

以 `tree_id = 7003`、`member_id = 1018001` 为例：

```sql
WITH spouse_info AS (
    SELECT 'spouse' AS relation_type, m2.member_id, m2.name, m2.gender, m2.birth_date
    FROM marriage mg
    JOIN member m2
      ON m2.tree_id = mg.tree_id
     AND (
          (mg.member_id_1 = 1018018 AND m2.member_id = mg.member_id_2)
       OR (mg.member_id_2 = 1018018 AND m2.member_id = mg.member_id_1)
     )
    WHERE mg.tree_id = 7003
),
child_info AS (
    SELECT 'child' AS relation_type, c.member_id, c.name, c.gender, c.birth_date
    FROM parent_child pc
    JOIN member c
      ON c.tree_id = pc.tree_id
     AND c.member_id = pc.child_member_id
    WHERE pc.tree_id = 7003
      AND pc.parent_member_id = 1018001
)
SELECT * FROM spouse_info
UNION ALL
SELECT * FROM child_info;
```

### 4.2 祖先递归查询

```sql
WITH RECURSIVE ancestor_cte AS (
    SELECT
        pc.tree_id,
        pc.parent_member_id AS ancestor_member_id,
        pc.child_member_id AS source_member_id,
        pc.parent_role,
        1 AS depth,
        ARRAY[pc.child_member_id, pc.parent_member_id]::bigint[] AS path_member_ids
    FROM parent_child pc
    WHERE pc.tree_id = 7003
      AND pc.child_member_id = 1018018

    UNION ALL

    SELECT
        pc.tree_id,
        pc.parent_member_id,
        a.source_member_id,
        pc.parent_role,
        a.depth + 1,
        a.path_member_ids || pc.parent_member_id
    FROM ancestor_cte a
    JOIN parent_child pc
      ON pc.tree_id = a.tree_id
     AND pc.child_member_id = a.ancestor_member_id
    WHERE a.depth < 12
)
SELECT
    a.depth,
    a.parent_role,
    m.member_id,
    m.name,
    m.gender,
    m.generation_no
FROM ancestor_cte a
JOIN member m
  ON m.tree_id = a.tree_id
 AND m.member_id = a.ancestor_member_id
ORDER BY a.path_member_ids;
```

### 4.3 平均寿命最长的一代

```sql
WITH generation_life AS (
    SELECT
        m.generation_no,
        AVG(EXTRACT(YEAR FROM age(COALESCE(m.death_date, CURRENT_DATE), m.birth_date))) AS avg_lifespan_years
    FROM member m
    WHERE m.tree_id = 7003
      AND m.birth_date IS NOT NULL
    GROUP BY m.generation_no
)
SELECT generation_no, avg_lifespan_years
FROM generation_life
ORDER BY avg_lifespan_years DESC, generation_no ASC
LIMIT 1;
```

### 4.4 大于 50 岁且无配偶的男性成员

```sql
SELECT
    m.member_id,
    m.name,
    EXTRACT(YEAR FROM age(CURRENT_DATE, m.birth_date))::int AS age_years,
    m.generation_no
FROM member m
WHERE m.tree_id = 7005
  AND m.gender = 'male'
  AND m.birth_date IS NOT NULL
  AND EXTRACT(YEAR FROM age(CURRENT_DATE, m.birth_date)) > 50
  AND NOT EXISTS (
      SELECT 1
      FROM marriage mg
      WHERE mg.tree_id = m.tree_id
        AND (mg.member_id_1 = m.member_id OR mg.member_id_2 = m.member_id)
  )
ORDER BY m.birth_date ASC, m.member_id ASC;
```

### 4.5 早于本代平均出生年份的成员

```sql
WITH generation_avg AS (
    SELECT
        generation_no,
        AVG(EXTRACT(YEAR FROM birth_date)) AS avg_birth_year
    FROM member
    WHERE tree_id = 7003
      AND birth_date IS NOT NULL
      AND generation_no IS NOT NULL
    GROUP BY generation_no
)
SELECT
    m.member_id,
    m.name,
    m.generation_no,
    EXTRACT(YEAR FROM m.birth_date)::int AS birth_year,
    g.avg_birth_year
FROM member m
JOIN generation_avg g
  ON g.generation_no = m.generation_no
WHERE m.tree_id = 7003
  AND m.birth_date IS NOT NULL
  AND EXTRACT(YEAR FROM m.birth_date) < g.avg_birth_year
ORDER BY m.generation_no, m.birth_date;
```
